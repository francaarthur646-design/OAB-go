function allDisciplines(){

}

function renderChips(){
  const wrap = document.getElementById("disciplineChips");
  wrap.innerHTML = "";
  allDisciplines().forEach(disc=>{
    const evs = EVENTS.filter(e=>e.disc===disc);
    const doneCount = evs.filter(e=>done[e.id]).length;
    const c = DISC_COLORS[disc] || {bg:"var(--grey-soft)",fg:"var(--grey)"};
    const chip = document.createElement("span");
    chip.className = "chip";
    chip.style.background = c.bg;
    chip.style.color = c.fg;
    chip.style.borderColor = "transparent";
    chip.textContent = disc + " " + doneCount + "/" + evs.length;
    wrap.appendChild(chip);
  });
}

function renderProgress(){
  const total = EVENTS.length;
  const doneCount = EVENTS.filter(e=>done[e.id]).length;
  const pct = total ? Math.round((doneCount/total)*100) : 0;
  document.getElementById("progressPct").textContent = pct + "%";
  document.getElementById("progressLabel").textContent = doneCount + " de " + total + " atividades concluídas";
  document.getElementById("progressBar").style.width = pct + "%";
  renderChips();

function pad(n){return n<10?"0"+n:""+n;}

function renderCalendar(){
  document.getElementById("monthTitle").textContent = MONTHS[viewMonth] + " " + viewYear;
  const cal = document.getElementById("calendar");
  cal.innerHTML = "";
  WEEKDAYS.forEach(w=>{
    const el = document.createElement("div");
    el.className="wd"; el.textContent=w;
    cal.appendChild(el);
  });
  const firstDay = new Date(viewYear, viewMonth, 1).getDay();
  const daysInMonth = new Date(viewYear, viewMonth+1, 0).getDate();
  const todayStr = localToday();

  for(let i=0;i<firstDay;i++){
    const el = document.createElement("div");
    el.className="day empty";
    cal.appendChild(el);
  }
  for(let day=1; day<=daysInMonth; day++){
    const dateStr = viewYear+"-"+pad(viewMonth+1)+"-"+pad(day);
    const evs = eventsByDate(dateStr);
    const el = document.createElement("div");
    el.className = "day" + (dateStr===todayStr ? " today":"") + (dateStr===selectedDate ? " selected":"");
    const num = document.createElement("div");
    num.className="num"; num.textContent=day;
    el.appendChild(num);
    if(evs.length){
      const dotsWrap = document.createElement("div");
      dotsWrap.className="dots";
      const discsToday = Array.from(new Set(evs.map(e=>e.disc))).slice(0,5);
      discsToday.forEach(disc=>{
        const dot = document.createElement("span");
        dot.className="dot";
        dot.style.background = (DISC_COLORS[disc]||{fg:"var(--grey)"}).fg;
        dotsWrap.appendChild(dot);
      });
      el.appendChild(dotsWrap);
      const allDone = evs.every(e=>done[e.id]);
      if(allDone){
        const badge = document.createElement("div");
        badge.className="done-badge";
        badge.textContent="✓";
        el.appendChild(badge);
      }
    }
    el.addEventListener("click", ()=>{ selectedDate = dateStr; renderCalendar(); renderDayPanel(); });
    cal.appendChild(el);
  }
}

function renderDayPanel(){
  const panel = document.getElementById("dayPanel");
  const search = document.getElementById("search").value.trim().toLowerCase();
  const filterDisc = document.getElementById("filterDisciplina").value;

  let evs = eventsByDate(selectedDate);
  if(filterDisc) evs = evs.filter(e=>e.disc===filterDisc);
  if(search) evs = evs.filter(e =>
    e.title.toLowerCase().includes(search) ||
    (e.prof||"").toLowerCase().includes(search) ||
    e.disc.toLowerCase().includes(search)
  );

  const dateObj = new Date(selectedDate+"T00:00:00");
  const label = dateObj.toLocaleDateString("pt-BR", {weekday:'long', day:'numeric', month:'long'});

  panel.innerHTML = "";
  const h2 = document.createElement("h2");
  const durMin=x=>{const m=/^(\d+)h(\d+)?$/.exec(x);return m?(+m[1])*60+(+(m[2]||0)):parseInt(x)||0;};
  const totMin=eventsByDate(selectedDate).filter(e=>e.disc!=="Simulado"&&e.disc!=="Mentoria"&&e.dur!=="1m").reduce((a,e)=>a+durMin(e.dur),0);
  h2.textContent = label.charAt(0).toUpperCase()+label.slice(1)+(totMin?" · "+Math.floor(totMin/60)+"h"+String(totMin%60).padStart(2,"0"):" · descanso");
  panel.appendChild(h2);

  if(!evs.length){
    const empty = document.createElement("div");
    empty.className="empty-state";
    empty.textContent = "Nenhuma atividade encontrada para este dia.";
    panel.appendChild(empty);
    return;
  }

  evs.sort((a,b)=>a.t.localeCompare(b.t));
  evs.forEach(e=>{
    const isDone = !!done[e.id];
    const c = DISC_COLORS[e.disc] || {bg:"var(--grey-soft)",fg:"var(--grey)"};
    const row = document.createElement("div");
    row.className = "event" + (isDone?" done":"");
    const cb = document.createElement("input");
    cb.type="checkbox"; cb.checked = isDone;
    cb.addEventListener("change", ()=>{
      if(cb.checked){ done[e.id]=true; } else { delete done[e.id]; }
      saveDone(); renderProgress(); renderCalendar(); renderDayPanel();
    });
    row.appendChild(cb);
    const body = document.createElement("div");
    body.className="event-body";
    const title = document.createElement("div");
    title.className="event-title";
    title.textContent = e.title;
    body.appendChild(title);
    const meta = document.createElement("div");
    meta.className="event-meta";
    const tag = document.createElement("span");
    tag.className="tag";
    tag.style.background=c.bg; tag.style.color=c.fg;
    tag.textContent = e.disc;
    meta.appendChild(tag);
    const metaText = document.createElement("span");
    metaText.className="meta-text";
    let txt = e.t + " · " + e.dur;
    if(e.prof) txt += " · " + e.prof;
    if(e.tipo) txt += " · " + e.tipo;
    metaText.textContent = txt;
    meta.appendChild(metaText);
    body.appendChild(meta);
    row.appendChild(body);
    panel.appendChild(row);
  });
}
